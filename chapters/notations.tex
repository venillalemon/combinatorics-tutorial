\chapter*{记号表}
\markboth{记号表}{记号表}

\begin{center}
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{>{\centering\arraybackslash}p{0.28\textwidth}p{0.58\textwidth}}
\toprule
记号 & 含义 \\
\midrule
$G=(V,E)$ & 图 $G$，其中 $V$ 是顶点集，$E$ 是边集 \\
$V(G)$ & 图 $G$ 的顶点集 \\
$E(G)$ & 图 $G$ 的边集 \\
$|G|$ 或 $n$ & 图 $G$ 的阶数，即顶点数 $|V(G)|$ \\
$m$ 或 $|E(G)|$ & 图 $G$ 的边数 \\
$N(v)$ & 顶点 $v$ 的邻居集合 \\
$\deg v$ & 顶点 $v$ 的度数 \\
$\Delta(G)$ & 图 $G$ 的最大度数 \\
$\delta(G)$ & 图 $G$ 的最小度数 \\
$\overline{d}(G)$ & 图 $G$ 的平均度数 \\
$\overline{G}$ & 图 $G$ 的补图 \\
$\omega(G)$ & 图 $G$ 的最大团的大小 \\
$\alpha(G)$ & 图 $G$ 的最大独立集的大小 \\
$\chi(G)$ & 图 $G$ 的最小染色数 \\
$K_n$ & $n$ 阶完全图 \\
$C_n$ & $n$ 阶圈 \\
$P_n$ & $n$ 阶路 \\
$T(n,p)$ & $n$ 阶完全 $p$ 部 Turan 图 \\
$t(n,p)$ & $T(n,p)$ 的边数 \\
$r(s,t)$ & Ramsey 数 \\
$\operatorname{Cr}(G)$ & 图 $G$ 的 crossing number \\
\bottomrule
\end{tabular}
\end{center}
